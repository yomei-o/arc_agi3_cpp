# Private build of phys_dag (does not touch the shared rdkit-min CMakeLists).
cmake_minimum_required(VERSION 3.20)
project(phys_dag CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_subdirectory(C:/prog/casmi/fwdmodel/rdkit-min rdkitmin)
add_executable(phys_dag C:/prog/casmi/fwdmodel/rdkit-min/phys_dag.cpp)
target_include_directories(phys_dag PRIVATE C:/prog/casmi/fwdmodel/rdkit-min)
target_link_libraries(phys_dag PRIVATE SmilesParse GraphMol autograd)
